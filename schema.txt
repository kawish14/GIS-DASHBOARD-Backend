-- This file is the reference schema -- create the database from it directly.
--
-- username and password_hash are nullable because an invited user has neither
-- until they complete registration; the CHECK constraints below are what stop
-- an account becoming active without them.
CREATE TABLE public.dashboard_users (
	id serial4 NOT NULL,
	username varchar(50) NULL,
	password_hash text NULL,
	full_name varchar(100) NULL,
	email varchar(100) NULL,
	role_id int4 NOT NULL,
	created_at timestamp DEFAULT now() NULL,
	current_session_id varchar(255) NULL,
	last_login_device text NULL,
	last_login_at timestamp NULL,
	status varchar(20) DEFAULT 'active' NOT NULL,   -- invited | active | disabled
	invited_at timestamp NULL,
	invited_by int4 NULL,
	activated_at timestamp NULL,
	invite_expires_at timestamp NULL,
	signup_token_hash text NULL,                    -- SHA-256 of the step-1 token
	signup_token_expires_at timestamp NULL,
	reset_token_hash text NULL,                     -- SHA-256 of the password-reset token
	reset_token_expires_at timestamp NULL,
	CONSTRAINT dashboard_users_pkey PRIMARY KEY (id),
	CONSTRAINT dashboard_users_username_key UNIQUE (username),
	CONSTRAINT dashboard_users_status_check CHECK (status IN ('invited', 'active', 'disabled')),
	CONSTRAINT dashboard_users_active_has_credentials CHECK (status <> 'active' OR (username IS NOT NULL AND password_hash IS NOT NULL)),
	CONSTRAINT dashboard_users_invited_has_email CHECK (status <> 'invited' OR email IS NOT NULL),
	CONSTRAINT dashboard_users_role_id_fkey FOREIGN KEY (role_id) REFERENCES public.roles(id) ON DELETE RESTRICT,
	CONSTRAINT dashboard_users_invited_by_fkey FOREIGN KEY (invited_by) REFERENCES public.dashboard_users(id) ON DELETE SET NULL
);

-- Registration looks accounts up by email, so it has to identify exactly one.
CREATE UNIQUE INDEX dashboard_users_email_lower_key ON public.dashboard_users (lower(email)) WHERE email IS NOT NULL;
CREATE INDEX dashboard_users_status_idx ON public.dashboard_users (status) WHERE status <> 'active';



CREATE TABLE public.roles (
	id serial4 NOT NULL,
	role_name varchar(50) NOT NULL,
	description text NULL,
	permissions jsonb DEFAULT '{"layers": [], "canEdit": false, "regions": [], "features": {}, "canExport": false}'::jsonb NOT NULL,
	created_at timestamp DEFAULT now() NULL,
	CONSTRAINT roles_pkey PRIMARY KEY (id),
	CONSTRAINT roles_role_name_key UNIQUE (role_name)
);

CREATE TABLE public."session" (
	sid varchar NOT NULL,
	sess json NOT NULL,
	expire timestamptz(6) NOT NULL,
	CONSTRAINT session_pkey PRIMARY KEY (sid)
);
CREATE INDEX "IDX_session_expire" ON public.session USING btree (expire);


-- User activity log, written by utils/activityLog.js. One row per action:
-- sign-ins (successful and failed), sign-outs, registrations, password resets,
-- and every change made from the admin panel.
--
-- user_id is SET NULL when the account is deleted, and username is a copy taken
-- at the time, so a deleted user's history still says who it was. For a failed
-- sign-in under a name that does not exist, user_id is NULL and username is
-- what was typed.
--
-- details holds whatever is specific to the action, e.g.
--   {"reason": "invalid_credentials"}            on a failed login
--   {"target_user_id": 12, "target_username": "ali.raza", "new_status": "disabled"}
CREATE TABLE public.user_logs (
	id bigserial NOT NULL,
	created_at timestamptz DEFAULT now() NOT NULL,
	user_id int4 NULL,
	username varchar(100) NULL,
	action varchar(50) NOT NULL,                    -- see ACTIONS in utils/activityLog.js
	status varchar(10) DEFAULT 'success' NOT NULL,  -- success | failure
	ip_address varchar(64) NULL,
	device varchar(150) NULL,                       -- "Chrome on Windows"
	user_agent text NULL,
	details jsonb NULL,
	CONSTRAINT user_logs_pkey PRIMARY KEY (id),
	CONSTRAINT user_logs_status_check CHECK (status IN ('success', 'failure')),
	CONSTRAINT user_logs_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.dashboard_users(id) ON DELETE SET NULL
);

CREATE INDEX user_logs_created_at_idx ON public.user_logs (created_at DESC);
CREATE INDEX user_logs_user_id_idx ON public.user_logs (user_id, created_at DESC);
CREATE INDEX user_logs_action_idx ON public.user_logs (action, created_at DESC);
